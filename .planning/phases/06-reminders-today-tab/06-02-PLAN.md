Rewrite ContentView.swift with the two-gate pattern per D-01, D-03, and RESEARCH.md Pattern 5.

Key decisions:
- `@AppStorage("notificationPermissionShown") private var notificationPermissionShown = false` declared on ContentView itself (per RESEARCH.md Pattern 5 — synchronous UserDefaults, no async load needed)
- Gate order: onboardingComplete → notificationPermissionShown → TabView (per D-01)
- `NotificationPermissionView(onComplete: { notificationPermissionShown = true })` — callback sets the flag, ContentView branch switches automatically
- `TabView(selection: $appState.selectedTab)` binds tab selection to AppState (per D-20)
- Tab tags MUST be: Stack = 0, Today = 1, Schedule = 2 (per D-20: notification tap switches to Today = tab index 1)
- Keep the existing `.task { }` modifier unchanged — it bootstraps onboardingComplete from Core Data

```swift
import SwiftUI

struct ContentView: View {
    @Environment(AppState.self) private var appState
    @Environment(\.managedObjectContext) private var context
    @AppStorage("notificationPermissionShown") private var notificationPermissionShown = false

    var body: some View {
        @Bindable var appState = appState

        Group {
            if appState.onboardingComplete {
                if !notificationPermissionShown {
                    NotificationPermissionView(onComplete: {
                        notificationPermissionShown = true
                    })
                } else {
                    TabView(selection: $appState.selectedTab) {
                        StackListView()
                            .tabItem { Label("Stack", systemImage: "pills") }
                            .tag(0)

                        TodayView()
                            .tabItem { Label("Today", systemImage: "checkmark.circle") }
                            .tag(1)

                        ScheduleView()
                            .tabItem { Label("Schedule", systemImage: "calendar") }
                            .tag(2)
                    }
                    .tint(.aquaPrimary)
                }
            } else {
                OnboardingCoordinatorView()
            }
        }
        .task {
            let profile = UserProfileStore.fetchOrCreate(in: context)
            appState.onboardingComplete = profile.onboardingComplete
        }
    }
}
```
