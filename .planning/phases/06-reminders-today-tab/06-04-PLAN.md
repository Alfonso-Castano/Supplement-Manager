Architecture notes:
- `@Environment(ScheduleStore.self)` provides the schedule data (per D-06); do NOT add a new Core Data fetch for items (per D-07)
- `@Environment(\.managedObjectContext)` provides the context for DailyCheckOff writes
- `@FetchRequest` initialized in `init()` with today's midnight predicate — reacts to Core Data changes automatically (toggle → write → FetchRequest updates → completion summary updates)
- BlockRow is a private struct inside TodayView.swift — pass items, the isComplete/isPast state, the toggle binding, and the markComplete action as parameters
- Use `displayName(_ item.label)` from DesignSystem for user-visible item labels (slug → "Vitamin D3")
- Time window string: format `suggestedTime` using a DateFormatter with "h:mm a" style — use the first item's suggestedTime for the block start; block end can be omitted or use a fixed window from the block name. For simplicity, show only the suggested time (e.g. "7:00 AM"). The UI-SPEC shows "[Start time] – [End time]" but the start time is derivable from the first item's suggestedTime; end time requires UserProfile — to avoid a separate profile fetch, show only the start time as "Suggested: 7:00 AM" or derive from suggestedTime directly

Key implementation note on toggle binding: `isOn` must bind to a `Bool` that triggers the Core Data write. Use a computed binding:
```swift
Binding(
    get: { isComplete(block: blockName) },
    set: { newValue in
        if newValue { markComplete(timeBlock: blockName) }
    }
)
```
This way toggling ON writes the record; the FetchRequest observes the change and updates isComplete reactively.

The D-13 spec says "Mark complete" is a single toggle (not undo). So the toggle can only go from OFF → ON. The Binding set closure only acts when `newValue == true`.

```swift
import SwiftUI
import CoreData

struct TodayView: View {
    @Environment(ScheduleStore.self) private var scheduleStore
    @Environment(\.managedObjectContext) private var context

    @FetchRequest private var checkOffs: FetchedResults<DailyCheckOff>

    init() {
        let today = Calendar.current.startOfDay(for: Date())
        _checkOffs = FetchRequest(
            sortDescriptors: [],
            predicate: NSPredicate(format: "checkOffDate == %@", today as NSDate)
        )
    }

    private func isComplete(block: String) -> Bool {
        checkOffs.contains { $0.timeBlockValue == block && $0.completed }
    }

    private func isPast(block: String, items: [ScheduledItem]) -> Bool {
        guard let first = items.first else { return false }
        return first.suggestedTime < Date()
    }

    private func markComplete(timeBlock: String) {
        let today = Calendar.current.startOfDay(for: Date())
        let request = DailyCheckOff.fetchRequest()
        request.predicate = NSPredicate(
            format: "checkOffDate == %@ AND timeBlock == %@",
            today as NSDate, timeBlock
        )
        request.fetchLimit = 1
        let existing = (try? context.fetch(request))?.first
        let record = existing ?? DailyCheckOff(context: context)
        record.id = record.id ?? UUID()
        record.checkOffDate = today
        record.timeBlock = timeBlock
        record.completed = true
        record.completedAt = Date()
        try? context.save()
    }

    private var completedCount: Int {
        let schedule = scheduleStore.currentSchedule()
        var count = 0
        if !schedule.morning.isEmpty && isComplete(block: "morning") { count += 1 }
        if !schedule.afternoon.isEmpty && isComplete(block: "afternoon") { count += 1 }
        if !schedule.evening.isEmpty && isComplete(block: "evening") { count += 1 }
        return count
    }

    private var totalBlockCount: Int {
        let schedule = scheduleStore.currentSchedule()
        var count = 0
        if !schedule.morning.isEmpty { count += 1 }
        if !schedule.afternoon.isEmpty { count += 1 }
        if !schedule.evening.isEmpty { count += 1 }
        return count
    }

    var body: some View {
        NavigationStack {
            let schedule = scheduleStore.currentSchedule()
            if schedule.allItems().isEmpty {
                ContentUnavailableView(
                    "Nothing scheduled today",
                    systemImage: "checkmark.circle",
                    description: Text("Add items to your stack to generate a schedule.")
                )
                .navigationTitle("Today")
            } else {
                ScrollView {
                    VStack(spacing: 16) {
                        // Completion summary
                        HStack {
                            if completedCount == totalBlockCount && totalBlockCount > 0 {
                                Text("All done for today.")
                                    .font(.swFootnote)
                                    .foregroundStyle(.sagePrimary)
                            } else {
                                Text("\(completedCount) of \(totalBlockCount) blocks done today")
                                    .font(.swFootnote)
                                    .foregroundStyle(.textSecondary)
                            }
                            Spacer()
                        }
                        .padding(.horizontal, 16)
                        .padding(.top, 8)

                        // Block cards
                        if !schedule.morning.isEmpty {
                            BlockRow(
                                blockName: "Morning",
                                timeBlock: "morning",
                                items: schedule.morning,
                                isComplete: isComplete(block: "morning"),
                                isPast: isPast(block: "morning", items: schedule.morning),
                                toggleBinding: Binding(
                                    get: { isComplete(block: "morning") },
                                    set: { newValue in if newValue { markComplete(timeBlock: "morning") } }
                                )
                            )
                        }
                        if !schedule.afternoon.isEmpty {
                            BlockRow(
                                blockName: "Afternoon",
                                timeBlock: "afternoon",
                                items: schedule.afternoon,
                                isComplete: isComplete(block: "afternoon"),
                                isPast: isPast(block: "afternoon", items: schedule.afternoon),
                                toggleBinding: Binding(
                                    get: { isComplete(block: "afternoon") },
                                    set: { newValue in if newValue { markComplete(timeBlock: "afternoon") } }
                                )
                            )
                        }
                        if !schedule.evening.isEmpty {
                            BlockRow(
                                blockName: "Evening",
                                timeBlock: "evening",
                                items: schedule.evening,
                                isComplete: isComplete(block: "evening"),
                                isPast: isPast(block: "evening", items: schedule.evening),
                                toggleBinding: Binding(
                                    get: { isComplete(block: "evening") },
                                    set: { newValue in if newValue { markComplete(timeBlock: "evening") } }
                                )
                            )
                        }
                    }
                    .padding(.top, 8)
                    .padding(.bottom, 48)
                }
                .navigationTitle("Today")
                .background(Color.backgroundPage.ignoresSafeArea())
            }
        }
    }
}

private struct BlockRow: View {
    let blockName: String
    let timeBlock: String
    let items: [ScheduledItem]
    let isComplete: Bool
    let isPast: Bool
    @Binding var toggleBinding: Bool

    // Time formatter for suggestedTime display
    private var timeString: String {
        guard let first = items.first else { return "" }
        let formatter = DateFormatter()
        formatter.dateFormat = "h:mm a"
        return formatter.string(from: first.suggestedTime)
    }

    var body: some View {
        VStack(alignment: .leading, spacing: 0) {
            // Block header
            HStack {
                VStack(alignment: .leading, spacing: 2) {
                    Text(blockName)
                        .font(.swHeadline)
                        .foregroundStyle(isComplete ? Color.sagePrimary : Color.textPrimary)
                    if !timeString.isEmpty {
                        Text(timeString)
                            .font(.swCallout)
                            .foregroundStyle(.textSecondary)
                    }
                }
                Spacer()
                if isComplete {
                    Label("Completed", systemImage: "checkmark.circle.fill")
                        .font(.swFootnote)
                        .foregroundStyle(.positiveFG)
                        .padding(.horizontal, 8)
                        .padding(.vertical, 4)
                        .background(Color.positiveBG)
                        .clipShape(Capsule())
                }
            }
            .padding(.horizontal, 20)
            .padding(.top, 20)
            .padding(.bottom, 12)

            Divider()
                .padding(.horizontal, 16)

            // Item list
            VStack(alignment: .leading, spacing: 8) {
                ForEach(items) { item in
                    HStack(spacing: 8) {
                        Circle()
                            .fill(Color.aquaPrimary.opacity(0.25))
                            .frame(width: 6, height: 6)
                        Text(displayName(item.label))
                            .font(.swBody)
                            .foregroundStyle(.textPrimary)
                        if !item.dosage.isEmpty {
                            Text(item.dosage)
                                .font(.swCallout)
                                .foregroundStyle(.textSecondary)
                        }
                        Spacer()
                    }
                }
            }
            .padding(.horizontal, 20)
            .padding(.vertical, 12)

            Divider()
                .padding(.horizontal, 16)

            // Mark complete row
            HStack {
                Text("Mark complete")
                    .font(.swBody)
                    .foregroundStyle((isPast && !isComplete) ? Color.textTertiary : Color.textPrimary)
                Spacer()
                Toggle("", isOn: $toggleBinding)
                    .labelsHidden()
                    .tint(.aquaPrimary)
                    .disabled(isPast && !isComplete)
                    .animation(.easeIn(duration: 0.3), value: isComplete)
            }
            .padding(.horizontal, 20)
            .padding(.vertical, 16)
        }
        .background(Color.backgroundCard)
        .clipShape(RoundedRectangle(cornerRadius: 12))
        .padding(.horizontal, 16)
        .opacity((isPast && !isComplete) ? 0.4 : 1.0)
    }
}
```

Note on the BlockRow toggle binding declaration: use `@Binding var toggleBinding: Bool` and pass `toggleBinding:` at the call site. This gives the toggle a real SwiftUI Binding to animate correctly.
