Implementation rules:
- Use `var onComplete: () -> Void` (not @Binding, not @Environment) — ContentView passes the callback
- Do NOT use `.fullScreenCover` — this view IS the ContentView branch content (if/else gate)
- CTA taps `UNUserNotificationCenter.current().requestAuthorization(options: [.alert, .sound, .badge])` using async/await; calls `onComplete()` regardless of whether permission was granted or denied (per D-04)
- Use `try?` (not `try`) so permission denial does not propagate an error
- All colors use DesignSystem dot-syntax tokens (`.backgroundPage`, `.aquaPrimary`, `.textPrimary`, `.textSecondary`)
- All fonts use DesignSystem tokens (`.swTitle1`, `.swBody`, `.swHeadline`)
- Copy from UI-SPEC is verbatim — do not paraphrase
- BenefitRow is a private struct inside the same file
- No NavigationStack wrapper needed — this view is displayed directly as a ContentView branch

```swift
import SwiftUI
import UserNotifications

struct NotificationPermissionView: View {
    var onComplete: () -> Void

    var body: some View {
        ScrollView {
            VStack(spacing: 32) {
                Image(systemName: "bell.badge.fill")
                    .font(.system(size: 56))
                    .foregroundStyle(.aquaPrimary)
                    .padding(.top, 24)

                Text("Never Miss a Dose")
                    .font(.swTitle1)
                    .foregroundStyle(.textPrimary)
                    .multilineTextAlignment(.center)

                Text("Get a reminder at the right time for each block in your schedule.")
                    .font(.swBody)
                    .foregroundStyle(.textSecondary)
                    .multilineTextAlignment(.center)

                VStack(alignment: .leading, spacing: 16) {
                    BenefitRow(icon: "clock.fill",
                               text: "Timed to your morning, afternoon, and evening blocks")
                    BenefitRow(icon: "list.bullet",
                               text: "Each reminder lists your specific supplements")
                    BenefitRow(icon: "bell.slash",
                               text: "You control notifications any time in Settings")
                }
            }
            .padding(.horizontal, 16)
            .padding(.bottom, 48)
        }
        .background(Color.backgroundPage.ignoresSafeArea())
        .safeAreaInset(edge: .bottom) {
            VStack {
                Button("Enable Reminders") {
                    Task {
                        try? await UNUserNotificationCenter.current()
                            .requestAuthorization(options: [.alert, .sound, .badge])
                        onComplete()
                    }
                }
                .buttonStyle(.borderedProminent)
                .tint(.aquaPrimary)
                .font(.swHeadline)
                .frame(maxWidth: .infinity)
                .frame(minHeight: 50)
                .padding(.horizontal, 16)
            }
            .padding(.vertical, 16)
            .background(Color.backgroundPage)
        }
    }
}

private struct BenefitRow: View {
    let icon: String
    let text: String

    var body: some View {
        HStack(spacing: 12) {
            Image(systemName: icon)
                .foregroundStyle(.aquaPrimary)
                .frame(width: 24)
            Text(text)
                .font(.swBody)
                .foregroundStyle(.textPrimary)
        }
    }
}
```
