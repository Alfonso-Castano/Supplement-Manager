### Task 1 — `TimingProperties` struct

**File:** `StackWise/Schedule/TimingProperties.swift` (NEW)

Create a value type that wraps the `timing_properties` JSON blob from the `ingredients` table.

```swift
import Foundation

struct TimingProperties: Equatable {
    let requiredFood: Bool
    let requiresEmptyStomach: Bool
    let preferredTimeOfDay: String?   // "morning" | "afternoon" | "evening" | nil
    let avoidWithCaffeine: Bool

    static let none = TimingProperties(
        requiredFood: false,
        requiresEmptyStomach: false,
        preferredTimeOfDay: nil,
        avoidWithCaffeine: false
    )

    /// Decode from the `timing_properties` JSON blob stored in the `ingredients` table.
    /// Returns `.none` if the blob is absent or unparseable.
    static func decode(from jsonString: String?) -> TimingProperties {
        guard let json = jsonString,
              let data = json.data(using: .utf8),
              let dict = try? JSONSerialization.jsonObject(with: data) as? [String: Any]
        else { return .none }

        return TimingProperties(
            requiredFood: dict["required_food"] as? Bool ?? false,
            requiresEmptyStomach: dict["requires_empty_stomach"] as? Bool ?? false,
            preferredTimeOfDay: dict["preferred_time_of_day"] as? String,
            avoidWithCaffeine: dict["avoid_with_caffeine"] as? Bool ?? false
        )
    }
}
```

**Verification:** `TimingProperties.decode(from: nil) == .none`, `TimingProperties.decode(from: "{\"required_food\":true}").requiredFood == true`.

### Task 2 — `IngredientRepository.timingProperties(for:)`

**File:** `StackWise/Data/IngredientRepository.swift` (EDIT — add one method)

Add after the existing `search(_:limit:)` method:

```swift
/// Returns timing constraints for a single ingredient slug.
/// Returns `TimingProperties.none` if the slug is not in the DB or has no timing blob.
func timingProperties(for slug: String) -> TimingProperties {
    guard let db else { return .none }
    let sql = "SELECT timing_properties FROM ingredients WHERE canonical_name = ? LIMIT 1"
    var stmt: OpaquePointer?
    defer { sqlite3_finalize(stmt) }
    guard sqlite3_prepare_v2(db, sql, -1, &stmt, nil) == SQLITE_OK else { return .none }
    sqlite3_bind_text(stmt, 1, slug, -1, SQLITE_TRANSIENT)
    guard sqlite3_step(stmt) == SQLITE_ROW else { return .none }
    let blob: String? = (sqlite3_column_type(stmt, 0) != SQLITE_NULL)
        ? String(cString: sqlite3_column_text(stmt, 0)) : nil
    return TimingProperties.decode(from: blob)
}
```
