    Modify the private `StackItemRow` struct:

    ```swift
    private struct StackItemRow: View {
        @ObservedObject var item: StackItem
        @Environment(AnalysisStore.self) private var store

        var body: some View {
            HStack(alignment: .top) {
                VStack(alignment: .leading, spacing: 4) {
                    HStack(spacing: 8) {
                        Text(item.labelValue)
                            .font(.body.weight(.medium))

                        // Existing Phase 2 custom-item icon — unchanged
                        if !item.isInDatabase {
                            Label("Custom", systemImage: "exclamationmark.triangle.fill")
                                .labelStyle(.iconOnly)
                                .foregroundStyle(.orange)
                                .accessibilityLabel("Not in interaction database")
                        }

                        // NEW: severity badge from analysis
                        if let severity = severityForRow {
                            Image(systemName: severity == .critical
                                  ? "exclamationmark.octagon.fill"
                                  : "exclamationmark.circle.fill")
                                .foregroundStyle(severity == .critical ? .red : .orange)
                                .accessibilityLabel(severity == .critical
                                                    ? "Critical interaction present"
                                                    : "Advisory interaction present")
                        }
                    }
                    if let dosage = item.dosage, !dosage.isEmpty {
                        Text(dosage)
                            .font(.subheadline)
                            .foregroundStyle(.secondary)
                    }
                }
                Spacer()
                Text(item.itemTypeValue.capitalized)
                    .font(.caption.weight(.semibold))
                    .padding(.horizontal, 8)
                    .padding(.vertical, 4)
                    .background(badgeColor(for: item.itemTypeValue).opacity(0.15))
                    .foregroundStyle(badgeColor(for: item.itemTypeValue))
                    .clipShape(Capsule())
            }
            .padding(.vertical, 2)
        }

        /// Returns the worst severity for this item's slug, or nil if no pair includes it.
        /// Custom items (isInDatabase == false) return nil — the Phase 2 orange triangle handles their state.
        private var severityForRow: InteractionSeverity? {
            guard item.isInDatabase, let slug = item.value(forKey: "activeIngredient") as? String else { return nil }
            return store.currentResults().worstSeverity(forSlug: slug)
        }

        private func badgeColor(for itemType: String) -> Color {
            itemType == "drug" ? .red : .blue
        }
    }
    ```

    Notes:
    - `store.currentResults()` is cheap (cached) — calling it from every row is fine.
    - The computed property does NOT participate in synergy badges. Synergy is celebratory and might be added in a future polish phase; out of scope for Phase 3 row badges (D-31 covers grouping on the analysis screen; this plan only surfaces threat-level information on rows).
    - `@Environment(AnalysisStore.self)` automatically subscribes the row to the `@Observable` properties it reads (`results`). After `AnalysisStore.invalidate()` → first `currentResults()` recompute, the row re-renders with the new severity.
