    ```swift
    enum AnalysisEngine {
        /// Inputs:
        ///   activeItems — every StackItem the user has marked isActive == true
        ///   pairFetcher — closure provided by the caller (test injects a stub; production passes InteractionRepository.pairwise)
        /// Output:
        ///   AnalysisResults with the four groups populated per D-31 routing rules.
        static func analyze(
            activeItems: [StackItemView],
            pairFetcher: (Set<String>) -> [InteractionPair]
        ) -> AnalysisResults
    }

    /// Minimal projection of StackItem so the engine is decoupled from Core Data.
    /// Constructed by AnalysisStore from real NSManagedObject instances.
    struct StackItemView: Equatable {
        let label: String          // user-visible label
        let slug: String           // StackItem.activeIngredient (snake_case canonical)
        let isInDatabase: Bool
    }
    ```

    Implementation rules:

    - Build `slugs = Set(activeItems.filter { $0.isInDatabase }.map { $0.slug })`.
    - If `slugs.count < 2`, return `AnalysisResults(critical: [], advisory: [], synergy: [], notInDatabase: <labels of !isInDatabase items>)`.
    - Call `pairFetcher(slugs)` exactly once. Iterate the returned `[InteractionPair]`:
      - if `pair.interactionType == .synergy` → append to `synergy` (regardless of severity).
      - else if `pair.severity == .critical` → append to `critical`.
      - else → append to `advisory`.
    - `notInDatabase = activeItems.filter { !$0.isInDatabase }.map { $0.label }` — keep insertion order so the UI matches the stack list order.
    - Within each of the three pair groups, preserve the order returned by `pairFetcher` (the repository chose SQL ordering; engine does not re-sort).
    - Pure function. No mutable state. No imports beyond `Foundation`.

  <action>
    Create:

    ```swift
    import CoreData

    @Observable
    final class AnalysisStore {
        private let repo: InteractionRepository
        private let context: () -> NSManagedObjectContext  // lazy provider; closure form avoids capturing context lifetime

        private(set) var results: AnalysisResults?
        private(set) var dirty: Bool = true              // start dirty so first read triggers recompute

        init(repo: InteractionRepository, context: @escaping () -> NSManagedObjectContext) {
            self.repo = repo
            self.context = context
        }

        /// Marks the cache stale. Called after every stack mutation.
        func invalidate() {
            dirty = true
            results = nil
        }

        /// Returns the current results. Recomputes from the live stack if dirty.
        /// Cheap to call repeatedly — recompute only happens on dirty transitions.
        @discardableResult
        func currentResults() -> AnalysisResults {
            if let results, !dirty { return results }
            let fresh = recompute()
            results = fresh
            dirty = false
            return fresh
        }

        private func recompute() -> AnalysisResults {
            let ctx = context()
            let request = StackItem.fetchRequest()
            request.predicate = NSPredicate(format: "isActive == YES")
            request.sortDescriptors = [NSSortDescriptor(key: "sortOrder", ascending: true)]
            let items: [StackItem] = (try? ctx.fetch(request)) ?? []
            let views = items.map {
                StackItemView(
                    label: $0.value(forKey: "label") as? String ?? "",
                    slug: $0.value(forKey: "activeIngredient") as? String ?? "",
                    isInDatabase: ($0.value(forKey: "isInDatabase") as? Bool) ?? false
                )
            }
            return AnalysisEngine.analyze(activeItems: views, pairFetcher: repo.pairwise(for:))
        }
    }
    ```

    Notes:
    - Use KVC (`value(forKey:)`) for `StackItem` field access since the project does not (yet) have a typed `labelValue` / `slugValue` helper consistently. If Phase 2 already added typed accessors on the `StackItem` subclass for these names (e.g. `labelValue`), prefer those — read `StackItem+CoreDataClass.swift` once to confirm, and either way the behavior is the same.
    - The `context: () -> NSManagedObjectContext` closure lets the store be instantiated at app start with a deferred context binding — useful for tests that swap in an in-memory context.
  </action>
