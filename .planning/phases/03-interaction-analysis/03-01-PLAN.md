    ```swift
    enum InteractionSeverity: String { case advisory, critical }
    enum InteractionType: String {
        case absorptionCompetition = "absorption_competition"
        case timingConflict       = "timing_conflict"
        case unsafeCombination    = "unsafe_combination"
        case synergy
    }
    enum InteractionDirection: String { case bidirectional, directional }

    struct InteractionPair: Equatable, Identifiable {
        let id: Int64
        let compoundA: String          // snake_case slug
        let compoundB: String          // snake_case slug
        let compoundAType: String      // "supplement" | "drug"
        let compoundBType: String      // "supplement" | "drug"
        let interactionType: InteractionType
        let severity: InteractionSeverity
        let direction: InteractionDirection?
        let mechanism: String?
        let effect: String             // NOT NULL in DB
        let suggestedAction: String    // NOT NULL in DB
        let timingRelevant: Bool
        let timingWindowHours: Int?
        let citations: String          // raw text from DB
        let confidence: String         // "high" | "medium" | "low"
        let conflictNote: String?
    }

    @Observable
    final class InteractionRepository {
        private var db: OpaquePointer?

        init(bundle: Bundle = .main, resource: String = "stackwise", ext: String = "sqlite") { /* same pattern as IngredientRepository */ }
        deinit { if let db { sqlite3_close_v2(db) } }   // use _v2 (Phase 3 pattern)

        /// Returns every interaction row where BOTH compound_a and compound_b appear in `slugs`.
        /// Slug matching is exact and case-sensitive (DB slugs are normalized snake_case).
        /// Returns an empty array when slugs.count < 2 or db is nil.
        func pairwise(for slugs: Set<String>) -> [InteractionPair]
    }
    ```

    Implementation requirements:

    - `pairwise(for:)` short-circuits to `[]` when `slugs.count < 2` (no pair possible) or when `db == nil`.
    - SQL: build an `IN (?, ?, ?, ...)` placeholder list whose arity equals `slugs.count`. Bind each slug as TEXT with `SQLITE_TRANSIENT`. Query shape:

      ```sql
      SELECT id, compound_a, compound_b, compound_a_type, compound_b_type,
             interaction_type, severity, mechanism, direction, effect,
             suggested_action, timing_relevant, timing_window_hours,
             citations, confidence, conflict_note
        FROM interactions
       WHERE compound_a IN (<placeholders>)
         AND compound_b IN (<placeholders>);
      ```

      Bind the slugs into both INs (you can either bind the slugs twice — once per IN — or use a CTE; binding twice is simpler and avoids prepared-statement caching surprises). The CHECK constraint on `direction` means we don't need to filter on it; the IN/IN test naturally selects valid pairs. (See D-32 in CONTEXT for rationale.)

    - Use `sqlite3_finalize` via `defer`. Use `String(cString:)` for TEXT columns; guard nullable columns (`mechanism`, `direction`, `timing_window_hours`, `conflict_note`) by checking `sqlite3_column_type(stmt, i) != SQLITE_NULL`.
    - `timingRelevant` reads `sqlite3_column_int(stmt, ...) != 0`.
    - `interactionType` and `severity` are parsed via `InteractionType(rawValue:)` / `InteractionSeverity(rawValue:)`. If parsing fails (impossible per DB CHECK, but defensive), skip the row — do not crash.
    - File-level `private let SQLITE_TRANSIENT` shadowing is OK if it collides with the one in `IngredientRepository.swift`. Both are file-private, so they coexist.
