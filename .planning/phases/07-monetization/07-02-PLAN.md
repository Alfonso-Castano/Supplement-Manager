import StoreKit
import SwiftUI

// Context enum differentiates the two paywall surfaces:
// .onboarding — initial trial sign-up; CTA says "Start Free Trial"; calls onComplete after purchase
// .recovery   — expired/lapsed subscription recovery; CTA says "Start Subscription"; no onComplete needed
enum PaywallContext {
    case onboarding
    case recovery
}

struct PaywallView: View {
    let paywallContext: PaywallContext
    var onComplete: (() -> Void)? = nil

    @Environment(SubscriptionService.self) private var subscriptionService
    @Environment(AppState.self) private var appState
    @Environment(\.managedObjectContext) private var context

    // Annual pre-selected by default (D-04)
    @State private var selectedProductID: String = "com.stackwise.app.annual"

    var body: some View {
        ScrollView {
            VStack(spacing: 32) {

                // MARK: Zone 1 — Headline
                VStack(spacing: 8) {
                    Text(paywallContext == .onboarding
                         ? "Your personalized supplement plan"
                         : "Your trial has ended")
                        .font(.swTitle1)
                        .foregroundStyle(.textPrimary)
                        .multilineTextAlignment(.center)

                    Text(paywallContext == .onboarding
                         ? "14-day free trial, then choose your plan"
                         : "Subscribe to continue managing your supplements safely")
                        .font(.swBody)
                        .foregroundStyle(.textSecondary)
                        .multilineTextAlignment(.center)
                }

                // MARK: Zone 2 — Social proof
                VStack(spacing: 4) {
                    HStack(spacing: 4) {
                        ForEach(0..<5) { _ in
                            Image(systemName: "star.fill")
                                .foregroundStyle(.yellow)
                                .font(.swCaption)
                        }
                    }
                    Text("\"Finally know what I'm actually taking.\"")
                        .font(.swBody)
                        .foregroundStyle(.textPrimary)
                        .multilineTextAlignment(.center)
                    Text("— App Store reviewer")
                        .font(.swCaption)
                        .foregroundStyle(.textSecondary)
                }
                .padding(.horizontal, 8)

                // MARK: Zone 3 — Pricing cards (D-04)
                VStack(spacing: 12) {
                    // Annual card — pre-selected, "Best Value" badge
                    PricingCardView(
                        title: "Annual",
                        badge: "Best Value",
                        primaryPrice: "$2.92/mo",
                        secondaryPrice: "$35.00 billed annually",
                        isSelected: selectedProductID == "com.stackwise.app.annual"
                    ) {
                        selectedProductID = "com.stackwise.app.annual"
                    }

                    // Monthly card — price anchor
                    PricingCardView(
                        title: "Monthly",
                        badge: nil,
                        primaryPrice: "$4.99/mo",
                        secondaryPrice: nil,
                        isSelected: selectedProductID == "com.stackwise.app.monthly"
                    ) {
                        selectedProductID = "com.stackwise.app.monthly"
                    }
                }

                // Error state for restore failures (D-08)
                if let error = subscriptionService.restoreError {
                    Text(error)
                        .font(.swCaption)
                        .foregroundStyle(.red)
                        .multilineTextAlignment(.center)
                }

                Text("No commitment. Cancel anytime.")
                    .font(.swFootnote)
                    .foregroundStyle(.textSecondary)
            }
            .padding(24)
        }
        .background(Color.backgroundPage.ignoresSafeArea())
        .safeAreaInset(edge: .bottom) {
            VStack(spacing: 12) {
                // CTA button
                Button {
                    Task {
                        guard let product = subscriptionService.products.first(
                            where: { $0.productIdentifier == selectedProductID }
                        ) else { return }
                        do {
                            try await subscriptionService.purchase(
                                product,
                                appState: appState,
                                context: context
                            )
                            if paywallContext == .onboarding {
                                onComplete?()
                            }
                        } catch {
                            // Purchase errors (e.g. network) — StoreKit shows system UI
                        }
                    }
                } label: {
                    if subscriptionService.isLoading {
                        ProgressView()
                            .tint(.white)
                            .frame(maxWidth: .infinity)
                            .frame(minHeight: 50)
                    } else {
                        Text(paywallContext == .onboarding ? "Start Free Trial" : "Start Subscription")
                            .font(.swHeadline)
                            .frame(maxWidth: .infinity)
                            .frame(minHeight: 50)

                            // restoreError is set on subscriptionService directly
                        }
                    }
                } label: {
                    if subscriptionService.isRestoring {
                        ProgressView()
                            .tint(.aquaPrimary)
                    } else {
                        Text("Restore Purchases")
                            .font(.swBody)
                            .foregroundStyle(.aquaPrimary)
                    }
                }
                .disabled(subscriptionService.isLoading || subscriptionService.isRestoring)
            }
            .padding(.vertical, 16)
            .background(Color.backgroundPage)
        }
        .interactiveDismissDisabled(true)
    }
}

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| AppState.subscriptionStatus → ContentView gate | Gate rendering decision is driven by this value; must only be written by SubscriptionService.resolveAndPersist |
| User interaction → StoreKit | Purchase and restore calls cross into Apple's StoreKit system |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-07-02-01 | Elevation of Privilege | ContentView gate | mitigate | Gate reads `appState.subscriptionStatus` which is only written by `SubscriptionService.resolveAndPersist` after StoreKit verification — no user-writable path to this value |
| T-07-02-02 | Elevation of Privilege | PaywallView bypass | accept | PaywallView is rendered inline in ContentView Group body — there is no dismiss gesture or navigation escape when shown inline (not as sheet). `interactiveDismissDisabled(true)` covers the .fullScreenCover surface |
| T-07-02-03 | Denial of Service | PaywallView empty products | accept | Products array may be empty if StoreKit product load fails; CTA button guard `products.first(where:)` returns nil and purchase Task returns early — no crash |
| T-07-02-04 | Spoofing | PaywallContext.onboarding → completeOnboarding | mitigate | completeOnboarding() calls UserProfileStore.markOnboarded which guards `trialStartDate == nil` (D-10) — cannot be called twice to reset trial clock |
</threat_model>
