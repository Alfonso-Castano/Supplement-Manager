**Step B: Create StackWise/Services/SubscriptionService.swift**

Create directory `StackWise/Services/` if it does not exist, then create the file:

```swift
import CoreData
import StoreKit

@MainActor
@Observable
final class SubscriptionService {
    private(set) var products: [Product] = []
    private(set) var isLoading: Bool = false
    var isRestoring: Bool = false
    var restoreError: String? = nil

    private let productIDs: Set<String> = [
        "com.stackwise.app.monthly",
        "com.stackwise.app.annual"
    ]
    private var updatesTask: Task<Void, Never>?

    func configure(appState: AppState, context: NSManagedObjectContext) async {
        // Start Transaction.updates listener before any product fetch.
        // Guard prevents duplicate listeners if configure() is called more than once (D-06).
        if updatesTask == nil {
            updatesTask = Task.detached { [weak self] in
                for await update in Transaction.updates {
                    guard let self else { return }
                    if let transaction = try? update.payloadValue {
                        await self.resolveAndPersist(appState: appState, context: context)
                        await transaction.finish()
                    }
                }
            }
        }

        // Load products — non-fatal on failure (paywall shows cached copy text)
        do {
            let loaded = try await Product.products(for: productIDs)
            // Sort: monthly first (price anchor), annual second (pre-selected)
            products = loaded.sorted { $0.price < $1.price }
        } catch {
            // Non-fatal: product load failure does not block status resolution
        }

        await resolveAndPersist(appState: appState, context: context)
    }

    func purchase(
        _ product: Product,
        appState: AppState,
        context: NSManagedObjectContext
    ) async throws {
        isLoading = true
        defer { isLoading = false }

        let result = try await product.purchase()

        switch result {
        case .success(let verification):
            // Security: use try (not try?) — unverified transaction must never unlock the app
            let transaction = try verification.payloadValue
            await resolveAndPersist(appState: appState, context: context)
            await transaction.finish()
        case .userCancelled:
            break
        case .pending:
            break
        @unknown default:
            break
        }
    }

    func restorePurchases(
        appState: AppState,
        context: NSManagedObjectContext
    ) async throws {
        isRestoring = true
        defer { isRestoring = false }
        // D-08: AppStore.sync() is for Restore button only — never call on launch
        try await AppStore.sync()
        await resolveAndPersist(appState: appState, context: context)
        // AppState.subscriptionStatus is now updated; ContentView gate re-evaluates automatically
    }

    // MARK: — Internal

    func resolveAndPersist(appState: AppState, context: NSManagedObjectContext) async {
        let profile = UserProfileStore.fetchOrCreate(in: context)
        let status = await resolveSubscriptionStatus(profile: profile)
        appState.subscriptionStatus = status
        UserProfileStore.updateSubscriptionStatus(status, profile: profile, in: context)
    }

    private func resolveSubscriptionStatus(profile: UserProfile) async -> String {
        // D-10: trialStartDate is nil before onboarding completes → fail open (D-11)
        guard let trialStart = profile.trialStartDate else { return "unknown" }

        let trialEnd = trialStart.addingTimeInterval(14 * 24 * 60 * 60)
        guard Date() >= trialEnd else { return "trial" }

        // Trial has elapsed — query StoreKit for active subscription
        // Transaction.currentEntitlements is a finite async sequence (terminates naturally)
        for await result in Transaction.currentEntitlements {
            if case .verified(let transaction) = result,
               transaction.productType == .autoRenewable,
               transaction.revocationDate == nil {
                return "active"
            }
        }
        return "expired"
    }

    deinit {
        updatesTask?.cancel()
    }
}
```

Key implementation notes:
- `resolveAndPersist` is `internal` (not `private`) so tests can call it directly
- `deinit` cancels `updatesTask` to prevent listener leak
- `try verification.payloadValue` (not `try?`) — per security constraint in RESEARCH.md §Security Domain
- `products.sorted { $0.price < $1.price }` puts monthly ($4.99) at index 0, annual ($35) at index 1

```swift
static func updateSubscriptionStatus(
    _ status: String,
    profile: UserProfile,
    in context: NSManagedObjectContext
) {
    guard profile.subscriptionStatus != status else { return }
    profile.subscriptionStatus = status
    do {
        try context.save()
    } catch {
        assertionFailure("Failed to update subscriptionStatus: \(error)")
    }
}
```

Pattern notes:
- `static func` — enum has no instance
- `in context: NSManagedObjectContext` — labeled parameter always last, matches `markOnboarded`
- Guard for no-op prevents unnecessary Core Data saves (profile already at same status)
- `assertionFailure` not `fatalError` — matches existing save error handling pattern

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-07-01-01 | Spoofing | SubscriptionService.purchase() | mitigate | `let transaction = try verification.payloadValue` — `try` (not `try?`) throws on invalid JWS; unverified transaction never reaches resolveAndPersist |
