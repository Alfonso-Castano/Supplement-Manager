<!-- Confidence decision tree (from RESEARCH.md): -->
<!--
NIH ODS / MedlinePlus / MSK source:
  + cites RCT / clinical trial → "high"
  + describes interaction effect (no trial) → "medium"
  + only mentions both compounds without describing effect → SKIP (explicit rejection rule)
OpenFDA source: always "low" (records from openfda go in interactions.json but NOT to SQLite)
PubMed source:
  + human clinical trial → "high"
  + observational human study or case report → "medium"
  + animal / in vitro / theoretical → "low"
-->

<!-- Interaction type mapping: -->
<!--
absorption_competition: one compound reduces absorption of the other
timing_conflict: must be taken at different times but both can be taken
unsafe_combination: should avoid combining; use template 3 or 7
synergy: well-paired compounds; positive interaction (use template 5)
-->

    ```python
    """
    Stage 2: Extraction
    Reads raw source files from research/raw/{nih_ods,medlineplus,msk}/
    Produces structured records in research/interactions.json.

    Extraction rules:
    - Explicit rejection rule: skip if no specific effect described between two compounds
    - Drug-drug exclusion: skip if both compounds are drugs
    - Lookup-first: resolve ALL compound names via resolve_canonical before writing
    - One record per pair: deduplication via pair_exists + merge_record (D-15)
    - Suggested action: always from action_templates.json; add new template to file if no fit
    - Atomic write: save_json_atomic after each record (D-08 crash safety)
    - timing_properties: write to canonical_names.json when source supports D-18 fields
    """
    import json
    import sys
    from datetime import date
    from pathlib import Path

    sys.path.insert(0, str(Path(__file__).parents[2]))
    from research.scripts.utils import (
        save_json_atomic, load_interactions, pair_exists, merge_record,
        resolve_canonical, load_canonical, load_checkpoint, save_checkpoint,
        RESEARCH_DIR, INTERACTIONS_PATH, CANONICAL_PATH
    )

    RAW_DIR = RESEARCH_DIR / "raw"
    TARGET_LIST_PATH = RESEARCH_DIR / "target_list.json"
    ACTION_TEMPLATES_PATH = RESEARCH_DIR / "action_templates.json"

    VALID_INTERACTION_TYPES = {"absorption_competition", "timing_conflict", "unsafe_combination", "synergy"}
    VALID_SEVERITIES = {"advisory", "critical"}
    VALID_CONFIDENCES = {"high", "medium", "low"}

    def load_target_types() -> dict[str, str]:
        """Returns {canonical_name: item_type} for all 70 targets."""
        with open(TARGET_LIST_PATH) as f:
            tl = json.load(f)
        result = {}
        for c in tl["supplements"]: result[c["canonical_name"]] = "supplement"
        for c in tl["medications"]: result[c["canonical_name"]] = "drug"
        return result

    def load_templates() -> list[dict]:
        with open(ACTION_TEMPLATES_PATH) as f:
            return json.load(f)["templates"]

    def validate_record(record: dict) -> list[str]:
        """Validate all required fields. Returns list of errors (empty = valid)."""
        errors = []
        required = ["compound_a", "compound_b", "compound_a_type", "compound_b_type",
                    "interaction_type", "severity", "effect", "suggested_action",
                    "timing_relevant", "citations", "confidence", "date_added"]
        for field in required:
            if field not in record or record[field] is None or record[field] == "":
                errors.append(f"Missing required field: {field}")

        if record.get("interaction_type") not in VALID_INTERACTION_TYPES:
            errors.append(f"Invalid interaction_type: {record.get('interaction_type')}")
        if record.get("severity") not in VALID_SEVERITIES:
            errors.append(f"Invalid severity: {record.get('severity')}")
        if record.get("confidence") not in VALID_CONFIDENCES:
            errors.append(f"Invalid confidence: {record.get('confidence')}")
        if record.get("compound_a_type") == "drug" and record.get("compound_b_type") == "drug":
            errors.append("Drug-drug record forbidden (both compound_a_type and compound_b_type = 'drug')")
        if not isinstance(record.get("citations"), list) or len(record["citations"]) == 0:
            errors.append("citations must be a non-empty list")

        return errors

    def write_record(record: dict, records: list[dict], canonical: dict) -> tuple[list[dict], str]:
        """
        Attempt to write a record to the records list.
        Returns (updated_records, action_taken) where action_taken is one of:
        'added', 'merged', 'skipped_validation', 'skipped_drug_drug', 'skipped_unresolved'
        """
        # Drug-drug guard
        if record.get("compound_a_type") == "drug" and record.get("compound_b_type") == "drug":
            return records, "skipped_drug_drug"

        # Resolve canonical names
        try:
            ca = resolve_canonical(record["compound_a"], canonical)
            cb = resolve_canonical(record["compound_b"], canonical)
            record["compound_a"] = ca
            record["compound_b"] = cb
        except LookupError as e:
            print(f"  UNRESOLVED CANONICAL: {e}")
            return records, "skipped_unresolved"

        # Validate
        errors = validate_record(record)
        if errors:
            print(f"  VALIDATION ERRORS for {record.get('compound_a')}/{record.get('compound_b')}: {errors}")
            return records, "skipped_validation"

        # Deduplication
        idx = pair_exists(records, record["compound_a"], record["compound_b"])
        if idx is not None:
            merge_record(records[idx], record)  # D-15 conflict resolution
            save_json_atomic(records, INTERACTIONS_PATH)
            return records, "merged"
        else:
            records.append(record)
            save_json_atomic(records, INTERACTIONS_PATH)
            return records, "added"
    ```
