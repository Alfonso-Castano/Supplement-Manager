    #!/usr/bin/env python3
    """
    Stage 1: Collection — NIH ODS, MedlinePlus, MSK
    Reads research/target_list.json, fetches source pages, saves to research/raw/.
    Fully resumable: already_collected() skips existing non-empty files.
    Rate limit: 1s between requests (government/academic resources).
    """
    import json
    import sys
    import time
    from pathlib import Path
    import requests
    from bs4 import BeautifulSoup

    # Add project root to sys.path so 'research.scripts.utils' resolves
    sys.path.insert(0, str(Path(__file__).parents[2]))
    from research.scripts.utils import already_collected, load_checkpoint, save_checkpoint, save_json_atomic

    RESEARCH_DIR = Path("research")
    RAW_DIR = RESEARCH_DIR / "raw"
    TARGET_LIST_PATH = RESEARCH_DIR / "target_list.json"

    HEADERS = {
        "User-Agent": "StackWise-Research/1.0 (supplement interaction database; contact: alfonsocastano486@gmail.com)"
    }

    def fetch_text(url: str, session: requests.Session) -> str | None:
        """Fetch a URL and return its plain text content, or None on error."""
        try:
            resp = session.get(url, headers=HEADERS, timeout=30)
            resp.raise_for_status()
            soup = BeautifulSoup(resp.text, "lxml")
            return soup.get_text(separator='\n').strip()
        except Exception as e:
            print(f"  ERROR fetching {url}: {e}")
            return None

    def save_raw(output_dir: Path, filename: str, url: str, content: str) -> None:
        """Save source URL as line 1, then content. Uses atomic write."""
        output_dir.mkdir(parents=True, exist_ok=True)
        path = output_dir / filename
        tmp = path.with_suffix(".tmp")
        with open(tmp, "w", encoding="utf-8") as f:
            f.write(url + "\n")
            f.write(content)
        import os; os.replace(tmp, path)

    def collect_nih_ods(target_list: dict, session: requests.Session) -> dict:
        """Fetch NIH ODS fact sheets for supplements that have nih_ods_url set."""
        out_dir = RAW_DIR / "nih_ods"
        results = {"collected": [], "skipped": [], "missing": [], "errors": []}

        for compound in target_list["supplements"]:
            name = compound["canonical_name"]
            url = compound.get("nih_ods_url")

            if not url:
                results["missing"].append(name)
                continue

            if already_collected(out_dir, name):
                results["skipped"].append(name)
                print(f"  SKIP (exists): nih_ods/{name}.txt")
                continue

            print(f"  Fetching NIH ODS: {name}")
            content = fetch_text(url, session)
            if content:
                save_raw(out_dir, f"{name}.txt", url, content)
                results["collected"].append(name)
            else:
                results["errors"].append(name)
            time.sleep(1)

        return results

    def collect_medlineplus(target_list: dict, session: requests.Session) -> dict:
        """Fetch MedlinePlus herb index and collect pages for target supplements."""
        out_dir = RAW_DIR / "medlineplus"
        results = {"collected": [], "skipped": [], "not_found": [], "errors": []}

        index_url = "https://medlineplus.gov/druginfo/herb_All.html"
        print(f"  Fetching MedlinePlus index: {index_url}")
        try:
            resp = session.get(index_url, headers=HEADERS, timeout=30)
            resp.raise_for_status()
            soup = BeautifulSoup(resp.text, "lxml")
        except Exception as e:
            print(f"  ERROR fetching MedlinePlus index: {e}")
            return results

        # Collect all herb links from the index
        herb_links = {}
        for a in soup.find_all("a", href=True):
            href = a["href"]
            if "/druginfo/natural/" in href or "/druginfo/herb_" in href.lower():
                text = a.get_text().strip().lower()
                if text:
                    herb_links[text] = href if href.startswith("http") else f"https://medlineplus.gov{href}"

        time.sleep(1)

        # Load canonical_names to resolve supplement names to index entries
        with open(RESEARCH_DIR / "canonical_names.json") as f:
            canonical = json.load(f)

        target_names = {c["canonical_name"] for c in target_list["supplements"]}

        for name in target_names:
            if already_collected(out_dir, name):
                results["skipped"].append(name)
                print(f"  SKIP (exists): medlineplus/{name}.txt")
                continue

            # Try to find the herb by canonical name or aliases
            all_names = [name.replace("_", " ")] + [a.replace("_", " ") for a in canonical.get(name, [])]
            found_url = None
            for variant in all_names:
                variant_lower = variant.lower()
                if variant_lower in herb_links:
                    found_url = herb_links[variant_lower]
                    break
                # Partial match
                for link_text, link_url in herb_links.items():
                    if variant_lower in link_text or link_text in variant_lower:
                        found_url = link_url
                        break
                if found_url:
                    break

            if not found_url:
                results["not_found"].append(name)
                continue

            print(f"  Fetching MedlinePlus: {name}")
            content = fetch_text(found_url, session)
            if content:
                save_raw(out_dir, f"{name}.txt", found_url, content)
                results["collected"].append(name)
            else:
                results["errors"].append(name)
            time.sleep(1)

        return results

    def collect_msk(target_list: dict, session: requests.Session) -> dict:
        """Crawl MSK About Herbs: paginate index, collect herb URLs, fetch each."""
        out_dir = RAW_DIR / "msk"
        results = {"collected": [], "skipped": [], "not_found": [], "errors": []}
        base = "https://www.mskcc.org"

        # Step 1: Collect all herb URLs from paginated index (pages 0-30)
        herb_urls: dict[str, str] = {}  # herb_name_slug -> full_url
        print("  Collecting MSK herb index (pages 0-30)...")
        for page in range(31):
            page_url = f"{base}/cancer-care/diagnosis-treatment/symptom-management/integrative-medicine/herbs/search?page={page}"
            try:
                resp = session.get(page_url, headers=HEADERS, timeout=30)
                resp.raise_for_status()
                soup = BeautifulSoup(resp.text, "lxml")
                for a in soup.find_all("a", href=True):
                    href = a["href"]
                    if "/cancer-care/integrative-medicine/herbs/" in href and href != "/cancer-care/integrative-medicine/herbs/search":
                        slug = href.rstrip("/").split("/")[-1]
                        full_url = href if href.startswith("http") else base + href
                        herb_urls[slug] = full_url
            except Exception as e:
                print(f"  ERROR fetching MSK page {page}: {e}")
            time.sleep(1)

        print(f"  MSK index collected: {len(herb_urls)} herb URLs found")

        # Load canonical_names for alias resolution
        with open(RESEARCH_DIR / "canonical_names.json") as f:
            canonical = json.load(f)

        target_names = {c["canonical_name"] for c in target_list["supplements"]}

        # Step 2: Match target compounds to collected herb URLs
        for name in target_names:
            if already_collected(out_dir, name):
                results["skipped"].append(name)
                print(f"  SKIP (exists): msk/{name}.txt")
                continue

            # Try canonical name and aliases as URL slugs
            all_variants = [name.replace("_", "-"), name.replace("_", " ")]
            all_variants += [a.replace(" ", "-").lower() for a in canonical.get(name, [])]

            found_url = None
            for variant in all_variants:
                if variant in herb_urls:
                    found_url = herb_urls[variant]
                    break
                # Partial slug match
                for slug, url in herb_urls.items():
                    if variant in slug or slug in variant:
                        found_url = url
                        break
                if found_url:
                    break

            if not found_url:
                results["not_found"].append(name)
                continue

            print(f"  Fetching MSK: {name}")
            content = fetch_text(found_url, session)
            if content:
                save_raw(out_dir, f"{name}.txt", found_url, content)
                results["collected"].append(name)
            else:
                results["errors"].append(name)
            time.sleep(1)

        return results

    if __name__ == "__main__":

    ```python
    #!/usr/bin/env python3
    """
    Stage 1: OpenFDA Adverse Events Collection
    Queries OpenFDA drug/event endpoint for each target supplement.
    All results get confidence='low' — used only as co-occurrence signals (D-09).
    Requires OPENFDA_API_KEY environment variable.
    """
    import json
    import os
    import sys
    import time
    from pathlib import Path
    import requests

    sys.path.insert(0, str(Path(__file__).parents[2]))
    from research.scripts.utils import load_checkpoint, save_checkpoint

    RESEARCH_DIR = Path("research")
    RAW_OPENFDA_DIR = RESEARCH_DIR / "raw" / "openfda"
    TARGET_LIST_PATH = RESEARCH_DIR / "target_list.json"
    CANONICAL_PATH = RESEARCH_DIR / "canonical_names.json"
    OPENFDA_BASE = "https://api.fda.gov/drug/event.json"

    HEADERS = {"User-Agent": "StackWise-Research/1.0 (supplement interaction database)"}

    def get_api_key() -> str:
        key = os.environ.get("OPENFDA_API_KEY", "")
        if not key:
            print("WARNING: OPENFDA_API_KEY not set. Using no-key limit (1,000 req/day).")
        return key

    def build_query_variants(canonical_name: str, aliases: list[str]) -> list[str]:
        """Build list of query strings for this supplement (canonical + top aliases).
        OpenFDA medicinalproduct values are user-reported — need multiple variants."""
        # Use canonical name (replace underscores with spaces, uppercase)
        variants = [canonical_name.replace("_", " ").upper()]
        # Add top 3 aliases (most common consumer names)
        for alias in aliases[:3]:
            variants.append(alias.upper())
        return list(dict.fromkeys(variants))  # deduplicate, preserve order

    def fetch_openfda(name_variant: str, api_key: str, session: requests.Session) -> list[dict]:
        """Query OpenFDA for a supplement name variant. Returns list of drug reports."""
        params = {
            "search": f'patient.drug.medicinalproduct:"{name_variant}"',
            "limit": 100,
        }
        if api_key:
            params["api_key"] = api_key

        try:
            resp = session.get(OPENFDA_BASE, params=params, headers=HEADERS, timeout=30)
            if resp.status_code == 404:
                return []  # No results for this query
            resp.raise_for_status()
            data = resp.json()
            return data.get("results", [])
        except Exception as e:
            print(f"  ERROR querying OpenFDA for '{name_variant}': {e}")
            return []

    def extract_co_occurrences(reports: list[dict]) -> list[dict]:
        """Extract multi-drug adverse event reports (co-occurrence signals).
        Only reports where patient.drug has 2+ entries are useful (D-05 co-occurrence detection)."""
        co_occurrences = []
        for report in reports:
            drugs = report.get("patient", {}).get("drug", [])
            if len(drugs) > 1:
                reactions = [r.get("reactionmeddrapt", "") for r in report.get("patient", {}).get("reaction", [])]
                co_occurrences.append({
                    "drugs": [d.get("medicinalproduct", "").upper() for d in drugs],
                    "reactions": reactions,
                    "report_id": report.get("safetyreportid", ""),
                })
        return co_occurrences

    def already_collected_openfda(canonical_name: str) -> bool:
        path = RAW_OPENFDA_DIR / f"{canonical_name}_openfda.json"
        return path.exists() and path.stat().st_size > 0

    if __name__ == "__main__":
        import os as _os
        RAW_OPENFDA_DIR.mkdir(parents=True, exist_ok=True)

        with open(TARGET_LIST_PATH) as f:
            target_list = json.load(f)
        with open(CANONICAL_PATH) as f:
            canonical = json.load(f)

        api_key = get_api_key()
        session = requests.Session()

        print("=== Stage 1: OpenFDA Collection ===")
        collected = []
        skipped = []
        errors = []

        for compound in target_list["supplements"]:
            name = compound["canonical_name"]

            if already_collected_openfda(name):
                skipped.append(name)
                print(f"  SKIP (exists): {name}_openfda.json")
                continue

            aliases = canonical.get(name, [])
            variants = build_query_variants(name, aliases)

            all_reports = []
            for variant in variants:
                print(f"  Querying OpenFDA: '{variant}'")
                reports = fetch_openfda(variant, api_key, session)
                all_reports.extend(reports)
                time.sleep(0.3)  # 240 req/min limit; 0.3s = ~200 req/min (safe margin)

            co_occurrences = extract_co_occurrences(all_reports)

            # Save: include query metadata + co_occurrences
            output = {
                "canonical_name": name,
                "query_variants": variants,
                "total_reports_fetched": len(all_reports),
                "co_occurrence_reports": len(co_occurrences),
                "co_occurrences": co_occurrences,
                "confidence": "low",  # D-09: all OpenFDA data is low confidence
                "note": "Co-occurrence signals only. Not a confirmed interaction source. Verify against NIH ODS, MedlinePlus, or PubMed before using.",
            }

            out_path = RAW_OPENFDA_DIR / f"{name}_openfda.json"
            tmp = out_path.with_suffix(".tmp")
            with open(tmp, "w", encoding="utf-8") as f:
                json.dump(output, f, indent=2)
            _os.replace(tmp, out_path)

            collected.append(name)

        print(f"\n=== OpenFDA Collection Complete ===")
        print(f"Collected: {len(collected)}")
        print(f"Skipped (existed): {len(skipped)}")
        print(f"Errors: {len(errors)}")
        print(f"Files in research/raw/openfda/: {len(list(RAW_OPENFDA_DIR.glob('*.json')))}")
    ```
