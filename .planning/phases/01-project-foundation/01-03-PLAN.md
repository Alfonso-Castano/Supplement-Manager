App entry point pattern (RESEARCH.md §Pattern 2):
```swift
@main
struct StackWiseApp: App {
    @State private var appState = AppState()    // @State owns @Observable — NOT @StateObject
    let persistenceController = PersistenceController.shared

    var body: some Scene {
        WindowGroup {
            ContentView()
                .environment(appState)                              // @Observable injection — NOT .environmentObject()
                .environment(\.managedObjectContext,
                    persistenceController.container.viewContext)    // Core Data injection
        }
    }
}
```

ContentView @Bindable pattern (RESEARCH.md §Pattern 2 — critical gotcha):
```swift
struct ContentView: View {
    @Environment(AppState.self) private var appState   // NOT @EnvironmentObject

    var body: some View {
        @Bindable var appState = appState   // MUST be inside body{} block, not as stored property
        TabView(selection: $appState.selectedTab) { ... }
    }
}
```

    **Step 1 — Create a solid-color 1024x1024 PNG app icon:**

    Using Python (available in the project environment), generate the icon programmatically:

    ```python
    # Run this in Terminal from the project root
    python3 -c "
    from PIL import Image
    img = Image.new('RGB', (1024, 1024), color=(0, 137, 123))
    img.save('StackWise/StackWise/Assets.xcassets/AppIcon.appiconset/AppIcon.png')
    print('Icon created')
    ```

    If PIL (Pillow) is not available, use this alternative with no dependencies:

    ```python
    python3 -c "
    import struct, zlib

    def create_png(width, height, r, g, b):
        def chunk(name, data):
            c = zlib.crc32(name + data) & 0xffffffff
            return struct.pack('>I', len(data)) + name + data + struct.pack('>I', c)
        ihdr = struct.pack('>IIBBBBB', width, height, 8, 2, 0, 0, 0)
        raw = b''
        for _ in range(height):
            raw += b'\\x00' + bytes([r, g, b] * width)
        idat = zlib.compress(raw)
        return b'\\x89PNG\\r\\n\\x1a\\n' + chunk(b'IHDR', ihdr) + chunk(b'IDAT', idat) + chunk(b'IEND', b'')

    with open('StackWise/StackWise/Assets.xcassets/AppIcon.appiconset/AppIcon.png', 'wb') as f:
        f.write(create_png(1024, 1024, 0, 137, 123))
    print('Teal icon created at 1024x1024')
    "
    ```
